"""Small CPU-only synthetic probes; never load checkpoints or audio files."""
from fractions import Fraction as F
from types import SimpleNamespace
import os

os.environ['OMP_NUM_THREADS'] = '1'
os.environ['MKL_NUM_THREADS'] = '1'

import numpy as np
import torch

from maichart.event_timing import EventPoint, EventTempo
from maichart.tempo_clock import TempoClock
from maichart.composer.stream_data import StreamStep, WAIT
import maichart.composer.stream_search as search
import maichart.composer.stream_quality as quality

torch.set_num_threads(1)
song = SimpleNamespace(tempo=[EventTempo(F(0), 120.)], first=0., audio=SimpleNamespace(duration_sec=5.))
clock = TempoClock(song.tempo, song.first)
original_gate, original_scorer, original_score = search._gate, quality.PhraseQualityScorer, quality.score_phrase
search._gate = lambda result, song, metrics, lifetimes: (SimpleNamespace(coverage={}), [])
quality.PhraseQualityScorer = lambda *args, **kwargs: object()
quality.score_phrase = lambda *args, **kwargs: {'total': .5, 'components': {}}

calls = []
def rollout(prefix, seed, target, deadline):
    start = prefix[-1].end if prefix else F(0)
    calls.append((str(start), str(target)))
    if len(calls) > 26:
        raise RuntimeError('bounded_probe_stopped')
    if start >= F(17,2):
        steps = prefix
        cursor, reason = F(10), 'learned_eos'
    else:
        steps = prefix + [StreamStep(WAIT, start, target, float(start)/2, float(target)/2)]
        cursor, reason = target, 'span_complete'
    return dict(steps=steps, points=[], cursor=cursor, reason=reason,
                retries=0, timing_retries=0, forced_waits=0,
                guard=SimpleNamespace(checks=0,rejections=0,reason_counts={}))

args = SimpleNamespace(phrase_candidates=1,phrase_write_beats=.5,phrase_commit_beats=.5,
    phrase_time_budget_sec=30.,phrase_repair_budget=0,phrase_rollback_budget=0,
    softcomfort_strength=0.,phrase_step_budget=16,seed=0)
try:
    search.search_stream(rollout, song, {}, args, F(10), clock, None)
except RuntimeError as exc:
    assert str(exc) == 'bounded_probe_stopped'
finally:
    search._gate, quality.PhraseQualityScorer, quality.score_phrase = original_gate, original_scorer, original_score

tail_calls = calls[-9:]
assert len(set(tail_calls)) == 1 and tail_calls[0] == ('17/2', '9')
print('G1 repeated_rollout_calls', len(calls), 'tail', tail_calls[:3])

seconds = torch.arange(0, 5.01, .01)
bank = torch.zeros((len(seconds), 238))
bank[:, 104+92] = torch.exp(-.5*((seconds-1.995)/.08)**2)
bank[:, 104+93] = bank[:,104+92]
bank[:, 104+94] = bank[:,104+92]
scorer = quality.PhraseQualityScorer(song, {'level':13.,'radar':{'note_density':.3}},
                                     audio_bank=(bank,seconds,seconds*2))
empty = scorer.score_phrase([], [], F(0), F(4), softcomfort_strength=0.)
boundary = scorer.score_phrase([EventPoint(F(4),'1')], [], F(0), F(4), softcomfort_strength=0.)
inside = scorer.score_phrase([EventPoint(F(399,100),'1')], [], F(0), F(4), softcomfort_strength=0.)
print('G2 empty',round(empty['total'],8),'boundary',round(boundary['total'],8),
      'inside',round(inside['total'],8),'boundary_nps',boundary['components']['actual_nps'],
      'inside_nps',inside['components']['actual_nps'])
assert boundary['total'] == empty['total'] and inside['total'] != boundary['total']
