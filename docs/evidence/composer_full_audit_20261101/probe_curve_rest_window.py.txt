"""Synthetic one-step timing support probe; no model/audio/checkpoint load."""
from fractions import Fraction as F
from types import SimpleNamespace
import os
os.environ['OMP_NUM_THREADS']='1'
os.environ['MKL_NUM_THREADS']='1'
import torch
from maichart.event_timing import EventTempo
from maichart.tempo_clock import TempoClock
from maichart.composer.stream_data import DeltaVocabulary, second_at_beat
from maichart.composer.stream_timing import timing_phases, current_timing_deltas
from maichart.composer.stream_sampling import note_delta_logits_for_audio

torch.set_num_threads(1)
vocab=DeltaVocabulary.fit([])
phases=timing_phases(vocab.values)
model=SimpleNamespace(config=SimpleNamespace(timing_output_policy='absolute_grid_v1'),
                      timing_grid=SimpleNamespace(phase_catalog=phases))
cursor=F(1,10)
clock=TempoClock([EventTempo(F(0),120.)],0.)
deltas=current_timing_deltas(model,None,cursor)
logits=note_delta_logits_for_audio(torch.zeros(len(deltas)),deltas,cursor,F(10),clock,5.,first_step=False)
legal=[cursor+d for d,value in zip(deltas,logits) if torch.isfinite(value)]
low=cursor+F(1,1000)
high=max(legal)+F(1,1000)
masked=[x for x in legal if low<=x<=high]
assert len(masked)==len(legal) and high<cursor+2 and high-low<2
print('absolute phases',len(phases),'legal candidates',len(legal),'rest beats',str(high-low),
      'unmasked',len(legal)-len(masked),'WAIT skip after rest beats',str(cursor+2-high))

model.config.timing_output_policy='relative_delta_v1'
model.config.delta_beats=vocab.values
deltas=current_timing_deltas(model,None,cursor)
logits=note_delta_logits_for_audio(torch.zeros(len(deltas)),deltas,cursor,F(10),clock,5.,first_step=False)
legal=[cursor+d for d,value in zip(deltas,logits) if torch.isfinite(value)]
assert cursor+2 in legal and cursor+2>high
print('relative legal delta2',cursor+2 in legal,'outside rest',cursor+2>high)
