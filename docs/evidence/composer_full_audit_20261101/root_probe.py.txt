"""Read-only production audit; all mutation confined to this scratch directory."""
import ast, hashlib, json, os
from pathlib import Path
from types import SimpleNamespace
from unittest.mock import patch

ROOT=Path(__file__).resolve().parents[2]
OUT=Path(__file__).parent
records=[]
for path in sorted((ROOT/'src').rglob('*')):
    if not path.is_file() or path.suffix not in ('.py','.js','.html','.css'):continue
    raw=path.read_bytes(); text=raw.decode('utf-8-sig')
    rec={'path':path.relative_to(ROOT).as_posix(),'lines':len(text.splitlines()),'sha256':hashlib.sha256(raw).hexdigest()}
    if path.suffix=='.py':
        tree=ast.parse(text);rec['functions']=sum(isinstance(n,(ast.FunctionDef,ast.AsyncFunctionDef)) for n in ast.walk(tree))
    records.append(rec)
(OUT/'source_inventory.json').write_text(json.dumps(records,ensure_ascii=False,indent=2),encoding='utf-8')

from maichart.webui.event_set_api import start_train
box=OUT/'busy_directory_probe';box.mkdir(exist_ok=True)
manifest=box/'manifest.json';manifest.write_text('{"songs":[]}',encoding='utf-8')
splitdir=box/'data_split';splitdir.mkdir(exist_ok=True)
victim=splitdir/'train_manifest.json';victim.write_text('ORIGINAL TRAIN SPLIT',encoding='utf-8')
events=[]
def split(source, output, **kwargs):
    events.append('split_write');victim.write_text('REPLACED BEFORE LOCK',encoding='utf-8')
    return {'splits':{'train':{'path':str(victim)},'dev':{'path':str(victim)}}}
def reserve(directory):
    events.append('reserve_rejected');raise ValueError('directory busy')
runtime=SimpleNamespace(_reserve_composer_training_task=reserve)
with patch('maichart.composer.evaluation.split_manifest',split):
    try:start_train(None,{'manifest':str(manifest),'checkpoint_dir':str(box),'mode':'fresh'},runtime=runtime)
    except Exception as e:result={'exception':str(e),'order':events,'split_after_rejection':victim.read_text(encoding='utf-8')}
(OUT/'busy_directory_probe.json').write_text(json.dumps(result,indent=2),encoding='utf-8')
print(json.dumps({'inventory_files':len(records),'inventory_lines':sum(r['lines'] for r in records),'busy_directory':result}))

logs=[]
for p in sorted((ROOT/'webui_workspace/logs').glob('*composer_train*.log'),key=lambda p:p.stat().st_mtime,reverse=True)[:3]:
    rows=[];runtime=None;recoveries=[];validation=[]
    for l in p.read_text(encoding='utf-8',errors='replace').splitlines():
        for tag,dest in [('composer_progress ',rows),('composer_recovery ',recoveries),('composer_validation ',validation)]:
            if tag in l:
                try:dest.append(json.loads(l.split(tag,1)[1]))
                except ValueError:pass
        if 'composer_runtime ' in l:
            runtime=json.loads(l.split('composer_runtime ',1)[1])
    logs.append({'path':str(p.relative_to(ROOT)),'runtime':runtime,'latest':rows[-1] if rows else None,'recoveries':recoveries,'validation':validation})
(OUT/'runtime_snapshot.json').write_text(json.dumps(logs,ensure_ascii=False,indent=2),encoding='utf-8')
